\Cref{alg:gd} is written with algorithm2e.

\begin{algorithm}[h]
  \caption{Gradient descent}\label{alg:gd}
  \KwIn{$x_0$, step size $\eta$}
  \KwOut{$x_T$}
  \For{$t \gets 0$ \KwTo $T-1$}{
    $x_{t+1} \gets x_t - \eta \nabla f(x_t)$\;
  }
  \eIf{$f(x_T) < f(x_0)$}{\Return{$x_T$}\;}{\Return{$x_0$}\;}
\end{algorithm}

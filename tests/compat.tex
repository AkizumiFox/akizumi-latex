\begin{problem}[二分搜尋]
在排序好的陣列裡找 $x$。
\begin{li}[label=(\roman*)]
  \item 寫出演算法。
  \item 分析時間複雜度。
\end{li}
\begin{algorithm}[h]
  \caption{Binary search}\label{alg:bs}
  \begin{algorithmic}[1]
    \Require sorted $a_1, \dots, a_n$, target $x$
    \State $\ell \gets 1$, $r \gets n$ \Comment{search $[\ell, r]$}
    \While{$\ell \le r$}
      \State $m \gets \lfloor (\ell + r)/2 \rfloor$
      \If{$a_m = x$} \State \Return $m$ \EndIf
    \EndWhile
  \end{algorithmic}
\end{algorithm}
\begin{solution}
見\cref{alg:bs}；\begin{CJK}{UTF8}{bkai}每一步範圍減半\end{CJK}，共 $O(\log n)$ 步。
\end{solution}
\end{problem}

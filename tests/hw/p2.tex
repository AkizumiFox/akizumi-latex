\begin{parts}[label=(\roman*)]
  \item Show that $(M, d)$ is complete.
  \item Find a convergent subsequence
  \begin{subparts}
    \item when $M$ is finite;
    \item when $M$ is infinite.
  \end{subparts}
\end{parts}
\begin{solution}
By \cref{prob:cauchy}, every Cauchy sequence converges.
\end{solution}
